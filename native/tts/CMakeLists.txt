cmake_minimum_required(VERSION 3.24)
project(LingroveSpeechRuntime LANGUAGES C CXX)

# TTS stays in a helper process: its newer libmtmd cannot replace the independently pinned
# completion library in the app process. Static linking also makes the helper relocatable.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE_DIR}" upstream EXCLUDE_FROM_ALL)
add_executable(lingrove-tts-helper SpeechHelper.cpp)
target_compile_features(lingrove-tts-helper PRIVATE cxx_std_17)
target_link_libraries(lingrove-tts-helper PRIVATE llama llama-common mtmd "-framework AudioToolbox")
install(TARGETS lingrove-tts-helper RUNTIME DESTINATION bin)
add_executable(lingrove-tts-decoder-check EXCLUDE_FROM_ALL DecoderCheck.cpp)
target_compile_features(lingrove-tts-decoder-check PRIVATE cxx_std_17)
target_link_libraries(lingrove-tts-decoder-check PRIVATE llama llama-common mtmd)
